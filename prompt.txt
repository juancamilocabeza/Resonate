<script>
(function(){
  function supportsMediaSession() {
    return 'mediaSession' in navigator;
  }

  function safe(fn) {
    try { fn(); } catch (e) { console.error('MediaSession error', e); }
  }

  // Initialize after all scripts have executed
  window.addEventListener('load', function initMediaSession() {
    if (!supportsMediaSession()) return;

    const ms = navigator.mediaSession;
    let positionUpdater = null;

    function getAudio() {
      return window.document ? document.getElementById('audio') : null;
    }

    function getCurrentTrackInfo() {
      // state and elements are defined later; when this runs they should exist.
      const s = window.state || {};
      const el = window.elements || {};
      const track = s.currentTrack || {};
      return {
        title: track.title || el.title?.textContent || 'Sin título',
        artist: track.creator || el.creator?.textContent || 'Desconocido',
        album: s.currentGenre ? (el.genreSelect?.options[el.genreSelect.selectedIndex]?.text || s.currentGenre) : undefined,
        id: track.id || null,
        url: track.url || null,
        artwork: track.id ? `https://archive.org/services/img/${track.id}` : null
      };
    }

    function updateMetadata() {
      safe(() => {
        const info = getCurrentTrackInfo();
        const artwork = info.artwork ? [
          { src: info.artwork, sizes: '512x512', type: 'image/jpeg' },
          { src: info.artwork + '/thumb', sizes: '96x96', type: 'image/jpeg' }
        ] : [];

        ms.metadata = new MediaMetadata({
          title: info.title,
          artist: info.artist,
          album: info.album || '',
          artwork: artwork
        });
      });
    }

    function updatePositionState() {
      safe(() => {
        const audio = getAudio();
        if (!audio || !ms.setPositionState) return;
        if (!isFinite(audio.duration) || audio.duration === 0) return;
        ms.setPositionState({
          duration: audio.duration,
          playbackRate: audio.playbackRate || 1,
          position: Math.max(0, audio.currentTime || 0)
        });
      });
    }

    function startPositionUpdater() {
      stopPositionUpdater();
      // update at a reasonable cadence
      positionUpdater = setInterval(updatePositionState, 1000);
      updatePositionState();
    }

    function stopPositionUpdater() {
      if (positionUpdater) {
        clearInterval(positionUpdater);
        positionUpdater = null;
      }
    }

    // Action handlers use existing global functions when possible.
    ms.setActionHandler('play', function() {
      safe(async () => {
        const audio = getAudio();
        if (audio && audio.paused) {
          // If no src, try to start the app/player
          if (!audio.src || audio.src === '') {
            if (typeof window.forcePlay === 'function') {
              await window.forcePlay();
            }
          } else {
            await audio.play().catch(()=>{});
          }
        }
        updateMetadata();
      });
    });

    ms.setActionHandler('pause', function() {
      safe(() => {
        const audio = getAudio();
        if (audio && !audio.paused) audio.pause();
      });
    });

    ms.setActionHandler('previoustrack', function() {
      safe(() => {
        // prefer existing function
        if (typeof window.playPrevious === 'function') {
          window.playPrevious();
        } else if (typeof window.playPreviousInContext === 'function') {
          window.playPreviousInContext();
        }
      });
    });

    ms.setActionHandler('nexttrack', function() {
      safe(() => {
        if (typeof window.playNext === 'function') {
          window.playNext();
        } else if (typeof window.playNextInContext === 'function') {
          window.playNextInContext();
        }
      });
    });

    ms.setActionHandler('seekbackward', function(details) {
      safe(() => {
        const audio = getAudio();
        if (!audio) return;
        const seekOffset = (details && details.seekOffset) || 10;
        audio.currentTime = Math.max(0, audio.currentTime - seekOffset);
        updatePositionState();
      });
    });

    ms.setActionHandler('seekforward', function(details) {
      safe(() => {
        const audio = getAudio();
        if (!audio) return;
        const seekOffset = (details && details.seekOffset) || 10;
        audio.currentTime = Math.min(audio.duration || Infinity, audio.currentTime + seekOffset);
        updatePositionState();
      });
    });

    ms.setActionHandler('seekto', function(details) {
      safe(() => {
        const audio = getAudio();
        if (!audio || typeof details.seekTime !== 'number') return;
        // If fastSeek available, use it
        if (typeof audio.fastSeek === 'function') {
          audio.fastSeek(details.seekTime);
        } else {
          audio.currentTime = details.seekTime;
        }
        updatePositionState();
      });
    });

    // Optional: stop action
    try {
      ms.setActionHandler('stop', function() {
        safe(() => {
          const audio = getAudio();
          if (audio) {
            audio.pause();
            audio.currentTime = 0;
          }
          stopPositionUpdater();
        });
      });
    } catch (e) {
      // some browsers don't support stop handler
    }

    // Keep MediaSession metadata and position state in sync with player events
    const audio = getAudio();
    if (audio) {
      audio.addEventListener('play', function() {
        updateMetadata();
        startPositionUpdater();
      });
      audio.addEventListener('pause', function() {
        updateMetadata();
        stopPositionUpdater();
        updatePositionState();
      });
      audio.addEventListener('seeked', updatePositionState);
      audio.addEventListener('ratechange', updatePositionState);
      audio.addEventListener('loadedmetadata', function(){
        updateMetadata();
        updatePositionState();
      });
      // When track changes via app code, try to update metadata: observe DOM mutations on title/creator/art
      const observer = new MutationObserver(function() {
        updateMetadata();
        updatePositionState();
      });
      const titleEl = document.getElementById('title');
      const creatorEl = document.getElementById('creator');
      if (titleEl) observer.observe(titleEl, { childList: true, characterData: true, subtree: true });
      if (creatorEl) observer.observe(creatorEl, { childList: true, characterData: true, subtree: true });
    }

    // Expose a helper so other code can request metadata update when it sets currentTrack.
    safe(() => {
      window.updateMediaSessionMetadata = function(trackId) {
        // When called, update metadata using current state + optional identifier
        if (trackId && (!getCurrentTrackInfo().id)) {
          // try to set a temporary artwork based on id
          const cur = getCurrentTrackInfo();
          cur.id = trackId;
        }
        updateMetadata();
        updatePositionState();
      };
    });

    // Initialize metadata once
    updateMetadata();
  }, { once: true });
})();
</script>

// ==================== SISTEMA DE CACHÉ MEJORADO ====================
const searchCache = new Map();
const metadataCache = new Map();

function getCacheKey(type, query) {
  return `${type}:${query.toLowerCase().trim()}`;
}

async function cachedSearch(query, searchFunction) {
  const cacheKey = getCacheKey('search', query);
  const cached = searchCache.get(cacheKey);
  
  if (cached && Date.now() - cached.timestamp < 300000) {
    return cached.data;
  }
  
  const results = await searchFunction(query);
  searchCache.set(cacheKey, {
    data: results,
    timestamp: Date.now()
  });
  
  return results;
}

// ==================== FETCH CON TIMEOUT MEJORADO ====================
async function fetchWithTimeout(resource, options = {}) {
  const { timeout = 15000, ...fetchOptions } = options;
  
  // Crear un AbortController para poder cancelar la petición
  const controller = new AbortController();
  const timeoutId = setTimeout(() => {
    controller.abort();
    console.log(`Fetch timeout after ${timeout}ms for: ${resource}`);
  }, timeout);
  
  try {
    const response = await fetch(resource, {
      ...fetchOptions,
      signal: controller.signal  
    });
    clearTimeout(timeoutId);
    return response;
  } catch (error) {
    clearTimeout(timeoutId);
    // Mejor manejo de errores de aborto
    if (error.name === 'AbortError') {
      throw new Error(`Timeout: La petición tardó más de ${timeout}ms`);
    }
    throw error;
  }
}

// ==================== VERIFICACIÓN DE URLS MEJORADA ====================
async function isUrlAccessible(url, timeout = 8000) {
  try {
    const response = await fetchWithTimeout(url, { 
      method: 'HEAD',
      timeout: timeout
    });
    return response.ok;
  } catch (error) {
    console.log('URL no accesible:', url, error.message);
    return false;
  }
}

// ==================== SISTEMA DE REPRODUCCIÓN CONTINUA ====================
class ContinuousPlaybackManager {
  constructor() {
    this.currentContext = null;
    this.contextData = {
      genre: { currentIndex: -1, items: [] },
      search: { currentIndex: -1, items: [] },
      favorites: { currentIndex: -1, items: [] },
      album: { currentIndex: -1, items: [], currentAlbumId: null }
    };
  }
  
  setContext(context, items, startIndex = 0, albumId = null) {
    this.currentContext = context;
    this.contextData[context] = {
      currentIndex: startIndex,
      items: items || [],
      currentAlbumId: albumId
    };
  }
  
  getNextTrack() {
    if (!this.currentContext || !this.contextData[this.currentContext].items.length) {
      return null;
    }
    
    const context = this.contextData[this.currentContext];
    
    // NUEVO: Lógica de shuffle para favoritos
    if (this.currentContext === 'favorites' && state.shuffleFavorites) {
      // Selección aleatoria para shuffle
      context.currentIndex = Math.floor(Math.random() * context.items.length);
    } else {
      // Comportamiento normal secuencial
      context.currentIndex = (context.currentIndex + 1) % context.items.length;
    }
    
    return context.items[context.currentIndex];
  }
  
  getCurrentContext() {
    return this.currentContext;
  }
  
  hasNextTrack() {
    const context = this.contextData[this.currentContext];
    return context && context.items.length > 0;
  }
  
  setAlbumPlaylist(albumId, tracks, startIndex = 0) {
    this.setContext('album', tracks, startIndex, albumId);
  }
  
  getCurrentAlbumId() {
    return this.contextData.album.currentAlbumId;
  }
  
  isPlayingAlbum(albumId = null) {
    if (albumId) {
      return this.currentContext === 'album' && this.contextData.album.currentAlbumId === albumId;
    }
    return this.currentContext === 'album';
  }
}

// Función mejorada para verificar si una pista específica es favorita
function isTrackFavorite(track) {
  if (!track || !track.url) return false;
  
  // Buscar por URL para identificar pistas específicas
  return state.favorites.some(fav => fav.url === track.url);
}

// Función mejorada para agregar favoritos con información de pista específica
function toggleFavorite(track) {
  if (!track || !track.url) return;
  
  const isFavorite = isTrackFavorite(track);
  
  if (isFavorite) {
    state.favorites = state.favorites.filter(fav => fav.url !== track.url);
    showStatus('Eliminado de favoritos', 'info');
  } else {
    const favoriteTrack = {
      id: track.id,
      title: track.title || 'Sin título',
      creator: track.creator || 'Desconocido',
      url: track.url,
      // Guardar información específica de la pista
      isAlbumTrack: track.isAlbumTrack || false,
      albumId: track.albumId || track.id,
      trackIndex: track.trackIndex || 0,
      totalTracks: track.totalTracks || 1,
      albumTitle: track.album || track.title
    };
    state.favorites.push(favoriteTrack);
    showStatus('Agregado a favoritos', 'info');
  }
  
  saveFavorites();
  updateFavoriteButton();
  updateFavoritesList();
}

function updateFavoriteButton() {
  if (state.currentTrack && state.currentTrack.url) {
    const isFavorite = isTrackFavorite(state.currentTrack);
    elements.favoriteBtn.classList.toggle('active', isFavorite);
    elements.favoriteBtn.textContent = isFavorite ? '❤️' : '🤍';
  } else {
    elements.favoriteBtn.classList.remove('active');
    elements.favoriteBtn.textContent = '🤍';
  }
}

function formatTime(seconds) {
  if (!isFinite(seconds) || seconds < 0) return '0:00';
  const minutes = Math.floor(seconds / 60);
  const remainingSeconds = Math.floor(seconds % 60);
  return `${minutes}:${String(remainingSeconds).padStart(2, '0')}`;
}

}

        messageEl.remove();
      }
    }, duration);
  }
}

// ==================== BÚSQUEDA OPTIMIZADA MEJORADA ====================
async function optimizedSearchItems(query, genre = null) {
  let searchQuery;
  if (genre && GENRE_CONFIG_OPTIMIZED[genre]) {
    const genreTerms = query || genre;
    searchQuery = `${genreTerms} AND mediatype:audio`;
  } else {
    searchQuery = `${query} AND mediatype:audio`;
  }
  
  const encodedQuery = encodeURIComponent(searchQuery);
  // MEJORA: Aumentar número de resultados y optimizar consulta
  const url = `https://archive.org/advancedsearch.php?q=${encodedQuery}&fl[]=identifier&fl[]=title&fl[]=creator&rows=50&output=json&sort[]=downloads+desc`;
  
  try {
    const response = await fetchWithTimeout(url, { timeout: 15000 });
    if (!response.ok) throw new Error('Error de red');
    const data = await response.json();
    
    const validItems = (data.response?.docs || []).filter(item => 
      item.identifier && item.title
    );
    
    console.log(`Encontrados ${validItems.length} items para: ${query}`);
    return validItems;
    
  } catch (error) {
    console.error('Error en búsqueda optimizada:', error);
    return [];
  }
}

// ==================== GESTIÓN DE ARCHIVOS DE AUDIO CORREGIDA ====================
async function fetchOptimizedPlayableFiles(identifier, genre = null) {
  const cacheKey = getCacheKey('metadata', identifier);
  const cached = metadataCache.get(cacheKey);
  
  if (cached && Date.now() - cached.timestamp < 600000) {
    return cached.data;
  }
  
  const url = `https://archive.org/metadata/${identifier}`;
  
  try {
    const response = await fetchWithTimeout(url, { timeout: 15000 });
    if (!response.ok) throw new Error('Error al cargar metadatos');
    const metadata = await response.json();
    
    if (!metadata?.files) {
      throw new Error('Metadatos inválidos');
    }
    
    // MEJORA: Filtrar y ordenar archivos por formato preferido (MP3 primero)
    const audioFiles = metadata.files.filter(file => 
      isPlayableAudioFile(file)
    ).sort((a, b) => {
      // Priorizar MP3 sobre otros formatos (generalmente más livianos)
      const aIsMp3 = a.name.toLowerCase().endsWith('.mp3');
      const bIsMp3 = b.name.toLowerCase().endsWith('.mp3');
      
      if (aIsMp3 && !bIsMp3) return -1;
      if (!aIsMp3 && bIsMp3) return 1;
      
      // Luego priorizar por tamaño (archivos más pequeños primero)
      return (a.size || 0) - (b.size || 0);
    });
    
    if (audioFiles.length === 0) {
      const fallbackAudioFiles = metadata.files.filter(file => 
        isPlayableAudioFileFallback(file)
      ).sort((a, b) => {
        const aIsMp3 = a.name.toLowerCase().endsWith('.mp3');
        const bIsMp3 = b.name.toLowerCase().endsWith('.mp3');
        if (aIsMp3 && !bIsMp3) return -1;
        if (!aIsMp3 && bIsMp3) return 1;
        return (a.size || 0) - (b.size || 0);
      });
      
      if (fallbackAudioFiles.length === 0) {
        throw new Error('No hay archivos de audio reproducibles');
      }
      
      console.log(`Usando archivos de respaldo: ${fallbackAudioFiles.length} archivos encontrados`);
      return createPlayableFiles(fallbackAudioFiles, identifier, metadata);
    }
    
    return createPlayableFiles(audioFiles, identifier, metadata);
    
  } catch (error) {
    console.error('Error al cargar archivos optimizados:', error);
    throw error;
  }
}

function createPlayableFiles(audioFiles, identifier, metadata) {
  let playableFiles;
  
  if (audioFiles.length > 1) {
    // Es un álbum - crear lista ordenada de pistas
    playableFiles = audioFiles
      .map((file, index) => {
        const trackInfo = extractTrackInfo(file, metadata.metadata);
        
        return {
          url: `https://archive.org/download/${identifier}/${encodeURIComponent(file.name)}`,
          title: trackInfo.title || metadata.metadata?.title || file.name || identifier,
          creator: trackInfo.artist || metadata.metadata?.creator || 'Artista desconocido',
          id: identifier,
          format: file.name.split('.').pop()?.toUpperCase() || 'MP3',
          trackNumber: trackInfo.trackNumber,
          duration: file.length || 0,
          album: metadata.metadata?.title || identifier,
          // Información adicional para identificar pistas de álbum
          isAlbumTrack: true,
          albumId: identifier,
          trackIndex: index,
          totalTracks: audioFiles.length
        };
      })
      .sort((a, b) => {
        // Ordenar por número de pista si está disponible
        if (a.trackNumber !== undefined && b.trackNumber !== undefined) {
          return a.trackNumber - b.trackNumber;
        }
        // Si no hay número de pista, ordenar alfabéticamente por título
        return (a.title || '').localeCompare(b.title || '');
      });
  } else {
    // Es una pista individual
    const audioFile = audioFiles[0];
    const trackInfo = extractTrackInfo(audioFile, metadata.metadata);
    
    playableFiles = [{
      url: `https://archive.org/download/${identifier}/${encodeURIComponent(audioFile.name)}`,
      title: trackInfo.title || metadata.metadata?.title || audioFile.name || identifier,
      creator: trackInfo.artist || metadata.metadata?.creator || 'Artista desconocido',
      id: identifier,
      format: audioFile.name.split('.').pop()?.toUpperCase() || 'MP3',
      album: metadata.metadata?.title || identifier,
      // Marcar como pista individual
      isAlbumTrack: false,
      trackIndex: 0,
      totalTracks: 1
    }];
  }
  
  // Guardar en caché
  metadataCache.set(getCacheKey('metadata', identifier), {
    data: playableFiles,
    timestamp: Date.now()
  });
  
  return playableFiles;
}

// MEJORA: Función extractTrackInfo mejorada para usar metadatos del archivo
function extractTrackInfo(file, itemMetadata) {
  const result = {
    title: null,
    artist: null,
    trackNumber: null
  };
  
  // Primero intentar usar los metadatos del archivo si están disponibles
  if (file.title) result.title = file.title;
  if (file.artist) result.artist = file.artist;
  if (file.creator) result.artist = file.creator;
  if (file.track && !isNaN(parseInt(file.track))) {
    result.trackNumber = parseInt(file.track);
  }
  
  // Si no hay título en los metadatos del archivo, intentar extraer del nombre
  if (!result.title) {
    const baseName = file.name.replace(/\.[^/.]+$/, ""); // Remover extensión
    
    // Patrones comunes en nombres de archivo de música
    const patterns = [
      // Formato: "01 - Canción.mp3", "01. Canción.mp3", "01 Canción.mp3"
      /^(\d+)[\s\.\-]+\s*(.+?)\s*$/,
      // Formato: "Artista - 01 - Canción.mp3"
      /^(.+?)\s*-\s*(\d+)[\s\.\-]+\s*(.+?)\s*$/,
      // Formato: "Artista - Canción.mp3"
      /^(.+?)\s*-\s*(.+?)\s*$/
    ];
    
    for (const pattern of patterns) {
      const match = baseName.match(pattern);
      if (match) {
        if (match.length === 3) {
          // Formato "01 - Canción"
          result.trackNumber = parseInt(match[1]);
          result.title = match[2].trim();
        } else if (match.length === 4) {
          // Formato "Artista - 01 - Canción"
          result.artist = match[1].trim();
          result.trackNumber = parseInt(match[2]);
          result.title = match[3].trim();
        } else if (match.length === 3 && pattern === patterns[2]) {
          // Formato "Artista - Canción"
          result.artist = match[1].trim();
          result.title = match[2].trim();
        }
        
        if (result.title) break;
      }
    }
    
    // Si no se pudo extraer del nombre, usar metadatos del item
    if (!result.title) {
      result.title = itemMetadata?.title || baseName;
    }
  }
  
  // Si no hay artista, usar metadatos del item
  if (!result.artist) {
    result.artist = itemMetadata?.creator;
  }
  
  return result;
}

// ==================== FILTROS DE ARCHIVOS DE AUDIO CORREGIDOS ====================
function isPlayableAudioFile(file) {
  if (!file.name) return false;
  
  // Excluir archivos de metadatos, comprimidos y otros no-audio
  const excludePatterns = [
    /_files\.xml$/i,
    /_meta\.xml$/i,
    /_reviews\.xml$/i,
    /_archive\.torrent$/i,
    /_meta\.sqlite$/i,
    /\.png$/i,
    /\.jpg$/i,
    /\.jpeg$/i,
    /\.gif$/i,
    /_thumb\./i,
    /spectrogram/i,
    /preview/i,
    /sample/i,
    /demo/i,
    /\.rar$/i,        // NUEVO: Excluir archivos RAR
    /\.zip$/i,        // NUEVO: Excluir archivos ZIP
    /\.7z$/i,         // NUEVO: Excluir archivos 7z
    /\.tar$/i,        // NUEVO: Excluir archivos TAR
    /\.gz$/i          // NUEVO: Excluir archivos GZ
  ];
  
  const hasExcludedPattern = excludePatterns.some(pattern => 
    pattern.test(file.name)
  );
  
  if (hasExcludedPattern) return false;
  
  // Incluir solo archivos de audio conocidos
  const audioFormats = ['.mp3', '.ogg', '.wav', '.flac', '.m4a', '.aac'];
  const isAudioFile = audioFormats.some(format => 
    file.name.toLowerCase().endsWith(format)
  );
  
  if (!isAudioFile) return false;
  
  // Verificar tamaño de archivo razonable
  const fileSize = file.size || 0;
  if (fileSize < 50000) return false; // Mínimo 50KB
  if (fileSize > 500000000) return false; // Máximo 500MB
  
  return true;
}

// Filtro de respaldo más permisivo
function isPlayableAudioFileFallback(file) {
  if (!file.name) return false;
  
  // Excluir solo los archivos obviamente no-audio
  const excludePatterns = [
    /_files\.xml$/i,
    /_meta\.xml$/i,
    /_reviews\.xml$/i,
    /\.png$/i,
    /\.jpg$/i,
    /\.jpeg$/i,
    /\.gif$/i,
    /\.rar$/i,        // NUEVO: Excluir archivos RAR
    /\.zip$/i,        // NUEVO: Excluir archivos ZIP
    /\.7z$/i,         // NUEVO: Excluir archivos 7z
    /\.tar$/i,        // NUEVO: Excluir archivos TAR
    /\.gz$/i          // NUEVO: Excluir archivos GZ
  ];
  
  const hasExcludedPattern = excludePatterns.some(pattern => 
    pattern.test(file.name)
  );
  
  if (hasExcludedPattern) return false;
  
  // Incluir archivos con extensiones de audio comunes
  const audioFormats = ['.mp3', '.ogg', '.wav', '.flac', '.m4a', '.aac'];
  const isAudioFile = audioFormats.some(format => 
    file.name.toLowerCase().endsWith(format)
  );
  
  return isAudioFile;
}

// ==================== FUNCIONES DE REPRODUCCIÓN OPTIMIZADAS ====================
async function playOptimizedRandom(genre) {
  if (state.isLoading) return;
  
  state.isLoading = true;
  elements.playBtn.disabled = true;
  elements.playBtn.innerHTML = '<span class="loader"></span>';
  
  state.currentGenre = genre;
  state.isPlayingFavorites = false;
  
  try {
    showStatus(`Buscando ${genre}...`, 'info', 0);
    
    const config = GENRE_CONFIG_OPTIMIZED[genre];
    if (!config) {
      throw new Error(`Configuración no encontrada para el género: ${genre}`);
    }
    
    const simpleQuery = `${genre}`;
    const items = await cachedSearch(
      `${genre} music`, 
      () => optimizedSearchItems(simpleQuery, genre)
    );
    
    if (items.length === 0) {
      const fallbackItems = await cachedSearch(
        `${genre}`,
        () => optimizedSearchItems(`${genre}`, null)
      );
      
      if (fallbackItems.length === 0) {
        throw new Error(`No se encontraron pistas para ${genre}. Intenta con otro género`);
      }
      
      playbackManager.setContext('genre', fallbackItems);
    } else {
      playbackManager.setContext('genre', items);
    }
    
    await playNextInContext();
    
  } catch (error) {
    console.error('Error en playOptimizedRandom:', error);
    handlePlaybackError(error, genre);
  } finally {
    state.isLoading = false;
    elements.playBtn.disabled = false;
    elements.playBtn.textContent = '❚❚';
  }
}

async function playOptimizedSearchResult(result, index, allResults) {
  if (state.isLoading) return;
  
  state.isLoading = true;
  elements.playBtn.disabled = true;
  
  try {
    showStatus(`Cargando: ${result.title}`, 'info', 0);
    
    const files = await fetchOptimizedPlayableFiles(result.identifier);
    
    if (files.length > 0) {
      if (files.length > 1) {
        // Es un álbum con múltiples pistas
        showStatus(`Álbum detectado: ${files.length} pistas`, 'info', 2000);
        // Guardar el origen del álbum como 'search'
        state.albumOrigin = 'search';
        playbackManager.setAlbumPlaylist(result.identifier, files, 0);
        
        // MEJORA: Verificar que la URL sea accesible antes de reproducir
        const firstTrack = files[0];
        if (await isUrlAccessible(firstTrack.url)) {
          await playAudioTrackOptimized(firstTrack, result.identifier, 0, files.length);
        } else {
          throw new Error('No se puede acceder al archivo de audio');
        }
      } else {
        // Pista individual
        // Para pistas individuales también establecer el origen
        state.albumOrigin = 'search';
        playbackManager.setContext('search', allResults, index);
        const track = files[0];
        
        // MEJORA: Verificar que la URL sea accesible antes de reproducir
        if (await isUrlAccessible(track.url)) {
          await playAudioTrackOptimized(track, result.identifier);
        } else {
          throw new Error('No se puede acceder al archivo de audio');
        }
      }
      hideSearchResults();
    } else {
      throw new Error('No se encontraron archivos de audio reproducibles');
    }
    
  } catch (error) {
    console.error('Error en playOptimizedSearchResult:', error);
    
    // MEJORA: Reintentar automáticamente con siguiente resultado
    if (allResults && index < allResults.length - 1) {
      showStatus(`Error con "${result.title}", intentando siguiente...`, 'error', 2000);
      setTimeout(() => {
        playOptimizedSearchResult(allResults[index + 1], index + 1, allResults);
      }, 1500);
    } else {
      showStatus(`Error: ${error.message}`, 'error');
    }
  } finally {
    state.isLoading = false;
    elements.playBtn.disabled = false;
  }
}

      trackIndex: favorite.trackIndex,

    };
    
    await playAudioTrackOptimized(track, favorite.id, favorite.trackIndex, favorite.totalTracks);

    updateShuffleButton();
    
  } catch (error) {

    showStatus(`Error: ${error.message}`, 'error');
  } finally {
    state.isLoading = false;
    elements.playBtn.disabled = false;

// ==================== REPRODUCCIÓN CONTINUA MEJORADA ====================
async function playNextInContext() {
  const context = playbackManager.getCurrentContext();
  
  // Verificar si estamos en el último track de un álbum de búsqueda
  if (context === 'album' && state.albumOrigin === 'search') {
    const albumContext = playbackManager.contextData.album;
    
    // Si es la última pista del álbum
    if (albumContext.currentIndex >= albumContext.items.length - 1) {
      console.log('Fin del álbum de búsqueda, continuando con siguiente resultado...');
      
      // Buscar el siguiente resultado de búsqueda
      const currentAlbumId = playbackManager.getCurrentAlbumId();
      const currentIndex = state.searchResults.findIndex(item => item.identifier === currentAlbumId);
      
      if (currentIndex !== -1 && currentIndex < state.searchResults.length - 1) {
        const nextResult = state.searchResults[currentIndex + 1];
        showStatus('Continuando con siguiente resultado de búsqueda...', 'info', 2000);
        
        // Limpiar el contexto actual de álbum
        playbackManager.setContext('search', state.searchResults, currentIndex + 1);
        
        // Reproducir el siguiente resultado
        await playOptimizedSearchResult(nextResult, currentIndex + 1, state.searchResults);
        return;
      } else {
        showStatus('Fin de los resultados de búsqueda', 'info', 2000);
        return;
      }
    }
  }
  
  // Comportamiento original para otros casos
  if (!playbackManager.hasNextTrack()) {
    if (context === 'genre') {
      await playOptimizedRandom(state.currentGenre);
    } else if (context === 'album') {
      showStatus('Fin del álbum', 'info', 2000);
      return;
    } else if (context === 'favorites') {
      // Si no hay más favoritos, reproducir Rock
      showStatus('Fin de los favoritos, reproduciendo Rock', 'info', 2000);
      await playOptimizedRandom('rock');
      return;
    }
    return;
  }
  
  const nextItem = playbackManager.getNextTrack();
  
  try {
    if (context === 'genre' || context === 'search') {
      const files = await fetchOptimizedPlayableFiles(nextItem.identifier, state.currentGenre);
      if (files && files.length > 0) {
        const track = files[0];
        await playAudioTrackOptimized(track, nextItem.identifier);
      } else {
        await playNextInContext();
      }
    } else if (context === 'favorites') {
      const track = {
        url: nextItem.url,
        title: nextItem.title,
        creator: nextItem.creator,
        id: nextItem.id
      };
      await playAudioTrackOptimized(track, nextItem.id);
    } else if (context === 'album') {
      // Reproducir siguiente pista del álbum actual
      const ctx = playbackManager.contextData.album;
      await playAudioTrackOptimized(nextItem, playbackManager.getCurrentAlbumId(), ctx.currentIndex, ctx.items.length);
    }
    
  } catch (error) {
    console.error('Error al reproducir siguiente pista:', error);
    setTimeout(() => playNextInContext(), 1000);
  }
}

async function playPreviousInContext() {
  const context = playbackManager.getCurrentContext();
  if (!context || !playbackManager.contextData[context].items.length) return;

  let ctx = playbackManager.contextData[context];
  ctx.currentIndex = (ctx.currentIndex - 1 + ctx.items.length) % ctx.items.length;
  const prevItem = ctx.items[ctx.currentIndex];

  try {
    if (context === 'genre' || context === 'search') {
      const files = await fetchOptimizedPlayableFiles(prevItem.identifier, state.currentGenre);
      if (files && files.length > 0) {
        const track = files[0];
        await playAudioTrackOptimized(track, prevItem.identifier);
      } else {
        await playPreviousInContext();
      }
    } else if (context === 'favorites') {
      const track = {
        url: prevItem.url,
        title: prevItem.title,
        creator: prevItem.creator,
        id: prevItem.id
      };
      await playAudioTrackOptimized(track, prevItem.id);
    } else if (context === 'album') {
      // Reproducir pista anterior del álbum actual
      await playAudioTrackOptimized(prevItem, playbackManager.getCurrentAlbumId(), ctx.currentIndex, ctx.items.length);
    }
  } catch (error) {
    console.error('Error al reproducir pista anterior:', error);
    setTimeout(() => playPreviousInContext(), 1000);
  }
}

async function playAudioTrackOptimized(track, identifier, trackIndex = 0, totalTracks = 1) {
  if (!track || !track.url) {
    throw new Error('Pista no disponible');
  }
  
  elements.audio.pause();
  elements.audio.currentTime = 0;
  elements.audio.src = '';
  
  // Ocultar favoritos y resultados cuando se reproduce una pista
  hideSearchResults();
  hideFavoritesList();
  
  // Mostrar información de álbum si hay múltiples pistas
  if (totalTracks > 1) {
    elements.title.textContent = `${track.title || 'Sin título'}`;
    elements.creator.textContent = `${track.creator || 'Desconocido'} (${trackIndex + 1}/${totalTracks})`;
  } else {
    elements.title.textContent = track.title;
    elements.creator.textContent = track.creator;
  }
  
  await setArtwork(track, identifier);
  
  // Guardar la pista actual con su URL completa
  state.currentTrack = {
    ...track,
    url: track.url,
    id: track.id || identifier,
    trackIndex: trackIndex,
    totalTracks: totalTracks,
    // Marcar si es una pista de álbum
    isAlbumTrack: totalTracks > 1,
    albumId: identifier
  };
  
  // Actualizar MediaSession
  if (typeof window.updateMediaSessionMetadata === 'function') {
    try { window.updateMediaSessionMetadata(state.currentTrack.id); } catch(e){ }
  }
  
  updateFavoriteButton();
  
  showStatus(`Cargando: ${track.title}`, 'info', 0);
  elements.playBtn.innerHTML = '<span class="loader"></span>';

  return new Promise((resolve, reject) => {
    const audio = elements.audio;
    let timeoutId;
    
    const onCanPlay = () => {
      cleanup();
      audio.play().then(() => {
        // Mensaje mejorado para álbumes
        if (totalTracks > 1) {
          showStatus(`Reproduciendo: ${track.title} (${trackIndex + 1}/${totalTracks})`, 'info');
        } else {
          showStatus(`Reproduciendo: ${track.title}`, 'info');
        }
        resolve(true);
      }).catch(reject);
    };
    
    const onError = (e) => {
      cleanup();
      reject(new Error(`Error de audio: ${e.target.error?.message || 'Fuente no compatible'}`));
    };
    
    const onStalled = () => {
      console.log('Audio stalled, waiting...');
      // No rechazamos inmediatamente, damos más tiempo
    };
    
    const cleanup = () => {
      audio.removeEventListener('canplay', onCanPlay);
      audio.removeEventListener('error', onError);
      audio.removeEventListener('stalled', onStalled);
      clearTimeout(timeoutId);
    };
    
    // MEJORA: Timeout más largo y con reintentos
    timeoutId = setTimeout(() => {
      // Si el audio no se ha cargado pero no hay error, intentar forzar play
      if (audio.readyState < 2) { // HAVE_CURRENT_DATA o menos
        console.log('Timeout - audio no listo, intentando play forzado');
        audio.play().then(() => {
          cleanup();
          resolve(true);
        }).catch(() => {
          cleanup();
          reject(new Error('Tiempo de espera agotado al cargar audio'));
        });
      } else {
        cleanup();
        reject(new Error('Tiempo de espera agotado al cargar audio'));
      }
    }, 25000); // Aumentado a 25 segundos
    
    audio.addEventListener('canplay', onCanPlay, { once: true });
    audio.addEventListener('error', onError, { once: true });
    audio.addEventListener('stalled', onStalled);
    
    audio.src = track.url;
    audio.load();
    
    // MEJORA: Precarga más agresiva
    audio.preload = 'auto';
  });
}

// ==================== MANEJO DE ERRORES MEJORADO ====================
function handlePlaybackError(error, context) {
  console.error(`Error de reproducción en ${context}:`, error);
  
  if (state.retryCount < state.maxRetries) {
    state.retryCount++;
    showStatus(`Reintentando... (${state.retryCount}/${state.maxRetries})`, 'info', 2000);
    
    setTimeout(() => {
      if (playbackManager.getCurrentContext() === 'genre') {
        playOptimizedRandom(state.currentGenre);
      } else {
        playNextInContext();
      }
    }, 2000);
  } else {
    showStatus(`Error: ${error.message}`, 'error');
    elements.title.textContent = 'Error al cargar';
    elements.creator.textContent = 'Intenta con otro género o busca manualmente';
    state.retryCount = 0;
    
    elements.playBtn.disabled = false;
    elements.playBtn.textContent = '▶';
  }
}

// ==================== SISTEMA DE BÚSQUEDA MEJORADO ====================
async function performOptimizedSearch() {
  const query = elements.searchInput.value.trim();
  
  if (!query) {
    hideSearchResults();
    return;
  }
  
  elements.searchResults.innerHTML = '<div class="search-loading"><span class="loader"></span>Buscando ...<div>';
  elements.searchResults.classList.add('active');
  state.searchActive = true;
  
  try {
    const results = await cachedSearch(query, () => optimizedSearchItems(query));
    
    // MEJORA: Enriquecer resultados con información de álbumes de forma más robusta
    const enrichedResults = [];
    
    // MEJORA: Procesar más resultados (hasta 30)
    for (let i = 0; i < Math.min(results.length, 30); i++) {
      const result = results[i];
      try {
        // Obtener metadatos rápidos para detectar álbumes con timeout más largo
        const metadataUrl = `https://archive.org/metadata/${result.identifier}`;
        const response = await fetchWithTimeout(metadataUrl, { timeout: 10000 });
        if (response.ok) {
          const metadata = await response.json();
          const audioFiles = metadata.files?.filter(file => 
            isPlayableAudioFile(file)
          ) || [];
          
          enrichedResults.push({
            ...result,
            isAlbum: audioFiles.length > 1,
            trackCount: audioFiles.length
          });
        } else {
          enrichedResults.push(result);
        }
      } catch (error) {
        console.log(`Error obteniendo metadatos para: ${result.identifier}`, error.message);
        // Si hay error, simplemente usar el resultado sin enriquecer
        enrichedResults.push(result);
      }
      
      // Pequeña pausa para no sobrecargar el servidor
      if (i < Math.min(results.length, 30) - 1) {
        await new Promise(resolve => setTimeout(resolve, 50));
      }
    }
    
    displaySearchResults(enrichedResults);
  } catch (error) {
    elements.searchResults.innerHTML = '<div class="no-results">Error en la búsqueda</div>';
  }
}

// ==================== NUEVA FUNCIÓN PARA MOSTRAR PISTAS DE ÁLBUM ====================
function displayAlbumTracks(albumId, tracksContainer, result) {
  // Limpiar contenedor
  tracksContainer.innerHTML = '<div class="search-loading"><span class="loader"></span>Cargando pistas...</div>';
  
  // Cargar y mostrar pistas del álbum
  fetchOptimizedPlayableFiles(albumId)
    .then(files => {
      tracksContainer.innerHTML = '';
      
      if (files.length === 0) {
        tracksContainer.innerHTML = '<div class="no-results">No se encontraron pistas</div>';
        return;
      }
      
      // Crear lista de pistas
      const tracksList = document.createElement('div');
      tracksList.className = 'tracks-list';
      
      files.forEach((track, index) => {
        const trackItem = document.createElement('div');
        trackItem.className = 'track-item';
        
        const isFavorite = isTrackFavorite(track);
        const trackNumber = track.trackNumber || index + 1;
        
        trackItem.innerHTML = `
          <div class="track-info">
            <div class="track-title">${track.title || 'Sin título'}</div>
            <div class="track-creator">${track.creator || 'Desconocido'}</div>
          </div>
          <div class="track-actions">
            <span class="track-number">${trackNumber}</span>
            <button class="track-play" data-url="${track.url}" data-title="${track.title}">▶</button>
            <button class="track-favorite ${isFavorite ? 'active' : ''}">${isFavorite ? '♥' : '♡'}</button>
          </div>
        `;
        
        // Evento para reproducir pista específica
        const playBtn = trackItem.querySelector('.track-play');
        playBtn.addEventListener('click', (e) => {
          e.stopPropagation();
          playSpecificTrack(track, albumId, index, files.length);
        });
        
        // Evento para favorito
        const favoriteBtn = trackItem.querySelector('.track-favorite');
        favoriteBtn.addEventListener('click', (e) => {
          e.stopPropagation();
          toggleFavorite(track);
          const isNowFavorite = isTrackFavorite(track);
          favoriteBtn.classList.toggle('active', isNowFavorite);
          favoriteBtn.textContent = isNowFavorite ? '♥' : '♡';
        });
        
        tracksList.appendChild(trackItem);
      });
      
      tracksContainer.appendChild(tracksList);
    })
    .catch(error => {
      console.error('Error cargando pistas del álbum:', error);
      tracksContainer.innerHTML = '<div class="no-results">Error al cargar las pistas</div>';
    });
}

function displaySearchResults(results) {
  elements.searchResults.innerHTML = '';
  state.searchResults = results;
  
  if (results.length === 0) {
    elements.searchResults.innerHTML = '<div class="no-results">No se encontraron resultados</div>';
    elements.searchResults.classList.add('active');
    return;
  }
  
  results.forEach((result, index) => {
    const resultEl = document.createElement('div');
    resultEl.className = 'search-result';
    
    // Verificar si alguna pista de este álbum está en favoritos
    const hasFavoriteTrack = state.favorites.some(fav => fav.albumId === result.identifier);
    
    resultEl.innerHTML = `
      <div class="result-info">
        <div class="result-title">
          ${result.title || 'Sin título'}
        </div>
        <div class="result-creator">${result.creator || 'Desconocido'}</div>
        ${result.isAlbum ? 
          `<div class="result-meta">
            <span class="result-album-indicator">🎵 Álbum: ${result.trackCount} pistas</span>
          </div>` : 
          ''}
      </div>
      <button class="result-favorite ${hasFavoriteTrack ? 'active' : ''}">
        ${hasFavoriteTrack ? '♥' : '♡'}
      </button>
    `;
    
    // Si es un álbum, agregar funcionalidad para expandir y mostrar pistas
    if (result.isAlbum) {
      const expandButton = document.createElement('button');
      expandButton.className = 'expand-tracks';
      expandButton.innerHTML = '▼ <span>Ver pistas</span>';
      resultEl.appendChild(expandButton);
      
      const tracksContainer = document.createElement('div');
      tracksContainer.className = 'tracks-container';
      tracksContainer.style.display = 'none';
      resultEl.appendChild(tracksContainer);
      
      expandButton.addEventListener('click', (e) => {
        e.stopPropagation();
        
        if (tracksContainer.style.display === 'none') {
          // Expandir
          displayAlbumTracks(result.identifier, tracksContainer, result);
          tracksContainer.style.display = 'block';
          expandButton.innerHTML = '▲ <span>Ocultar pistas</span>';
          state.expandedAlbums.add(result.identifier);
        } else {
          // Contraer
          tracksContainer.style.display = 'none';
          expandButton.innerHTML = '▼ <span>Ver pistas</span>';
          state.expandedAlbums.delete(result.identifier);
        }
      });
      
      // Si ya estaba expandido, expandirlo de nuevo
      if (state.expandedAlbums.has(result.identifier)) {
        expandButton.click();
      }
    }
    
    resultEl.addEventListener('click', async (e) => {
      if (!e.target.classList.contains('result-favorite') && 
          !e.target.classList.contains('expand-tracks') &&
          !e.target.closest('.expand-tracks') &&
          !e.target.classList.contains('track-play') &&
          !e.target.classList.contains('track-favorite')) {
        showStatus(`Cargando: ${result.title}`, 'info', 0);
        await playOptimizedSearchResult(result, index, results);
      }
    });
    
    const favoriteBtn = resultEl.querySelector('.result-favorite');
    favoriteBtn.addEventListener('click', async (e) => {
      e.stopPropagation();
      
      try {
        const files = await fetchOptimizedPlayableFiles(result.identifier);
        if (files.length > 0) {
          if (result.isAlbum) {
            // Para álbumes, agregar o quitar todas las pistas
            const hasAnyFavorite = state.favorites.some(fav => fav.albumId === result.identifier);
            
            if (hasAnyFavorite) {
              // Remover todas las pistas de este álbum
              state.favorites = state.favorites.filter(fav => fav.albumId !== result.identifier);
              showStatus('Álbum eliminado de favoritos', 'info');
              favoriteBtn.classList.remove('active');
              favoriteBtn.textContent = '♡';
            } else {
              // Agregar todas las pistas del álbum
              files.forEach(track => {
                if (!isTrackFavorite(track)) {
                  toggleFavorite(track);
                }
              });
              showStatus('Álbum agregado a favoritos', 'info');
              favoriteBtn.classList.add('active');
              favoriteBtn.textContent = '♥';
            }
          } else {
            // Para pistas individuales, agregar o quitar la primera pista
            const track = files[0];
            toggleFavorite(track);
            const isNowFavorite = isTrackFavorite(track);
            favoriteBtn.classList.toggle('active', isNowFavorite);
            favoriteBtn.textContent = isNowFavorite ? '♥' : '♡';
          }
        }
      } catch (error) {
        console.error('Error al obtener URL para favorito:', error);
        showStatus('Error al agregar favorito', 'error');
      }
    });
    
    elements.searchResults.appendChild(resultEl);
  });
  
  elements.searchResults.classList.add('active');
}

function setupRealTimeSearch() {
  let searchTimeout = null;
  
  elements.searchInput.addEventListener('input', () => {
    clearTimeout(searchTimeout);
    
    const query = elements.searchInput.value.trim();
    
    if (query.length < 2) {
      hideSearchResults();
      return;
    }
    
    // Mostrar estado de carga
    elements.searchResults.innerHTML = '<div class="search-loading"><span class="loader"></span>Buscando ...<div>';
    elements.searchResults.classList.add('active');
    state.searchActive = true;
    
    // Debounce para búsqueda en tiempo real
    searchTimeout = setTimeout(() => {
      performOptimizedSearch();
    }, 500);
  });
  
  // Búsqueda con Enter
  elements.searchInput.addEventListener('keydown', (e) => {
    if (e.key === 'Enter') {
      clearTimeout(searchTimeout);
      performOptimizedSearch();
    } else if (e.key === 'Escape') {
      hideSearchResults();
    }
  });
}

// ==================== GESTIÓN DE FAVORITOS MEJORADA ====================
function updateFavoritesList() {
  const favoritesList = elements.favoritesList;
  favoritesList.innerHTML = '';
  
  if (state.favorites.length === 0) {
    favoritesList.innerHTML = '<div class="no-results" style="padding: 12px; text-align: center;">No hay favoritos</div>';
    return;
  }
  
  // Agrupar favoritos por álbum
  const favoritesByAlbum = {};
  
  state.favorites.forEach(favorite => {
    const albumId = favorite.albumId || favorite.id;
    if (!favoritesByAlbum[albumId]) {
      favoritesByAlbum[albumId] = {
        albumTitle: favorite.albumTitle || favorite.title,
        albumCreator: favorite.creator,
        tracks: []
      };
    }
    favoritesByAlbum[albumId].tracks.push(favorite);
  });
  
  // Mostrar favoritos agrupados por álbum
  Object.keys(favoritesByAlbum).forEach(albumId => {
    const albumData = favoritesByAlbum[albumId];
    const isAlbum = albumData.tracks.length > 1;
    
    const albumItem = document.createElement('div');
    albumItem.className = 'favorite-item';
    
    // Mostrar indicador de álbum y número de pistas
    const albumIndicator = isAlbum ? 
      `<span class="album-indicator">Álbum: ${albumData.tracks.length} pistas</span>` : 
      '';
    
    albumItem.innerHTML = `
      <div class="favorite-info">
        <div class="favorite-title">
          ${albumData.albumTitle || 'Sin título'} ${albumIndicator}
        </div>
        <div class="favorite-creator">${albumData.albumCreator || 'Desconocido'}</div>
      </div>
      <div class="favorite-actions">
        <button class="favorite-play" data-id="${albumId}" data-is-album="${isAlbum}">▶</button>
        <button class="favorite-remove" data-id="${albumId}">×</button>
      </div>
    `;
    
    const playBtn = albumItem.querySelector('.favorite-play');
    const removeBtn = albumItem.querySelector('.favorite-remove');
    
    playBtn.addEventListener('click', (e) => {
      e.stopPropagation();
      const isAlbum = playBtn.getAttribute('data-is-album') === 'true';
      const id = playBtn.getAttribute('data-id');
      
      // Lógica para reproducir álbum completo o pista individual
      if (isAlbum) {
        playAlbumFromFavorites(id);
      } else {
        // Encontrar y reproducir la pista individual
        const favoriteTrack = state.favorites.find(fav => fav.albumId === id);
        if (favoriteTrack) {
          playOptimizedFavorite(state.favorites.indexOf(favoriteTrack));
        }
      }
    });
    
    removeBtn.addEventListener('click', (e) => {
      e.stopPropagation();
      const albumId = removeBtn.getAttribute('data-id');
      
      // Eliminar todas las pistas del álbum
      state.favorites = state.favorites.filter(fav => fav.albumId !== albumId);
      saveFavorites();
      updateFavoritesList();
      updateFavoriteButton();
      showStatus('Eliminado de favoritos', 'info');
    });
    
    favoritesList.appendChild(albumItem);
    
    // Mostrar pistas individuales si es un álbum con múltiples pistas
    if (isAlbum && albumData.tracks.length > 1) {
      const tracksContainer = document.createElement('div');
      tracksContainer.className = 'tracks-list';
      
      albumData.tracks.forEach((favorite, index) => {
        const trackItem = document.createElement('div');
        trackItem.className = 'track-item';
        
        trackItem.innerHTML = `
          <div class="track-info">
            <div class="track-title">${favorite.title || 'Sin título'}</div>
            <div class="track-creator">${favorite.creator || 'Desconocido'}</div>
          </div>
          <div class="track-actions">
            <span class="track-number">${favorite.trackIndex + 1}</span>
            <button class="track-play" data-url="${favorite.url}">▶</button>
            <button class="track-favorite active">♥</button>
          </div>
        `;
        
        const trackPlayBtn = trackItem.querySelector('.track-play');
        const trackFavoriteBtn = trackItem.querySelector('.track-favorite');
        
        trackPlayBtn.addEventListener('click', (e) => {
          e.stopPropagation();
          // Reproducir pista específica
          const favoriteIndex = state.favorites.findIndex(fav => fav.url === favorite.url);
          if (favoriteIndex !== -1) {
            playOptimizedFavorite(favoriteIndex);
          }
        });
        
        trackFavoriteBtn.addEventListener('click', (e) => {
          e.stopPropagation();
          // Eliminar solo esta pista
          state.favorites = state.favorites.filter(fav => fav.url !== favorite.url);
          saveFavorites();
          updateFavoritesList();
          updateFavoriteButton();
          showStatus('Pista eliminada de favoritos', 'info');
        });
        
        tracksContainer.appendChild(trackItem);
      });
      
      favoritesList.appendChild(tracksContainer);
    }
  });
}

// ==================== BARRA DE PROGRESO MEJORADA ====================
function updateProgressBar() {
  if (!state.isSeeking && elements.audio.duration && isFinite(elements.audio.duration)) {
    const progressPercent = (elements.audio.currentTime / elements.audio.duration) * 100;
    elements.progress.style.width = `${progressPercent}%`;
    elements.currentTime.textContent = formatTime(elements.audio.currentTime);
    elements.totalTime.textContent = formatTime(elements.audio.duration);
  }
}

function seekAudio(event) {
  if (!elements.audio.duration || !isFinite(elements.audio.duration)) return;
  
  const progressBarRect = elements.progressBar.getBoundingClientRect();
  const clickPosition = (event.clientX - progressBarRect.left) / progressBarRect.width;
  const seekTime = clickPosition * elements.audio.duration;
  
  elements.audio.currentTime = seekTime;
  updateProgressBar();
}

// NUEVO: Función mejorada para manejar el arrastre en la barra de progreso
function setupProgressBarInteraction() {
  let isDragging = false;
  
  const startDrag = (e) => {
    if (!elements.audio.duration || !isFinite(elements.audio.duration)) return;
    
    isDragging = true;
    state.isSeeking = true;
    elements.progressBar.classList.add('dragging');
    updateProgressOnDrag(e);
    
    // Prevenir selección de texto durante el arrastre
    e.preventDefault();
  };
  
  const drag = (e) => {
    if (!isDragging) return;
    updateProgressOnDrag(e);
  };
  
  const endDrag = () => {
    if (!isDragging) return;
    
    isDragging = false;
    state.isSeeking = false;
    elements.progressBar.classList.remove('dragging');
    
    // Asegurarse de que el audio continúe reproduciéndose si estaba reproduciendo
    if (!elements.audio.paused) {
      elements.audio.play().catch(() => {});
    }
  };
  
  const updateProgressOnDrag = (e) => {
    const progressBarRect = elements.progressBar.getBoundingClientRect();
    let clientX;
    
    if (e.type.includes('touch')) {
      clientX = e.touches[0].clientX;
    } else {
      clientX = e.clientX;
    }
    
    const clickPosition = Math.max(0, Math.min(1, (clientX - progressBarRect.left) / progressBarRect.width));
    const seekTime = clickPosition * elements.audio.duration;
    
    // Actualizar visualmente sin afectar la reproducción hasta que se suelte
    elements.progress.style.width = `${clickPosition * 100}%`;
    elements.currentTime.textContent = formatTime(seekTime);
    
    // Si es el evento final, establecer el tiempo real
    if (!isDragging) {
      elements.audio.currentTime = seekTime;
    }
  };
  
  // Eventos para mouse
  elements.progressBar.addEventListener('mousedown', startDrag);
  document.addEventListener('mousemove', drag);
  document.addEventListener('mouseup', endDrag);
  
  // Eventos para touch
  elements.progressBar.addEventListener('touchstart', startDrag);
  document.addEventListener('touchmove', drag);
  document.addEventListener('touchend', endDrag);
  
  // Evento de clic para progreso inmediato
  elements.progressBar.addEventListener('click', (e) => {
    if (!isDragging) {
      seekAudio(e);
    }
  });
}

// ==================== NUEVO: FUNCIONALIDAD DE CAST ====================
function setupCast() {
  // Verificar si la API de Cast está disponible
  if (!window.chrome || !window.chrome.cast || !window.chrome.cast.isAvailable) {
    console.log('Chromecast no disponible');
    return;
  }
  
  // Inicializar la API de Cast
  const context = cast.framework.CastContext.getInstance();
  
  // Configurar opciones
  const options = {
    receiverApplicationId: chrome.cast.media.DEFAULT_MEDIA_RECEIVER_APP_ID,
    autoJoinPolicy: chrome.cast.AutoJoinPolicy.ORIGIN_SCOPED
  };
  
  context.setOptions(options);
  
  // Escuchar cambios en el estado de Cast
  context.addEventListener(
    cast.framework.CastContextEventType.SESSION_STATE_CHANGED,
    (event) => {
      switch (event.sessionState) {
        case cast.framework.SessionState.SESSION_STARTED:
          console.log('Sesión de Cast iniciada');
          state.castSession = context.getCurrentSession();
          elements.castBtn.classList.add('active');
          showStatus('Conectado a dispositivo', 'info', 2000);
          break;
          
        case cast.framework.SessionState.SESSION_RESUMED:
          console.log('Sesión de Cast reanudada');
          state.castSession = context.getCurrentSession();
          elements.castBtn.classList.add('active');
          break;
          
        case cast.framework.SessionState.SESSION_ENDED:
          console.log('Sesión de Cast finalizada');
          state.castSession = null;
          elements.castBtn.classList.remove('active');
          showStatus('Desconectado del dispositivo', 'info', 2000);
          break;
      }
    }
  );
  
  // Escuchar cambios en el estado del receptor
  context.addEventListener(
    cast.framework.CastContextEventType.CAST_STATE_CHANGED,
    (event) => {
      switch (event.castState) {
        case cast.framework.CastState.NO_DEVICES_AVAILABLE:
          state.castAvailable = false;
          elements.castBtn.style.display = 'none';
          break;
          
        case cast.framework.CastState.NOT_CONNECTED:
          state.castAvailable = true;
          elements.castBtn.style.display = 'flex';
          elements.castBtn.classList.remove('active');
          break;
          
        case cast.framework.CastState.CONNECTING:
          state.castAvailable = true;
          elements.castBtn.classList.add('active');
          break;
          
        case cast.framework.CastState.CONNECTED:
          state.castAvailable = true;
          elements.castBtn.classList.add('active');
          break;
      }
    }
  );
  
  // Manejar clic en el botón de Cast
  elements.castBtn.addEventListener('click', () => {
    if (!state.castAvailable) {
      showStatus('No hay dispositivos disponibles', 'error', 2000);
      return;
    }
    
    const context = cast.framework.CastContext.getInstance();
    
    if (context.getSessionState() === cast.framework.SessionState.SESSION_STARTED) {
      // Si ya hay una sesión, finalizarla
      context.endCurrentSession(true);
    } else {
      // Iniciar nueva sesión
      context.requestSession().catch((err) => {
        console.error('Error al solicitar sesión de Cast:', err);
        showStatus('Error al conectar con el dispositivo', 'error', 2000);
      });
    }
  });
  
  // Función para enviar contenido a Cast
  window.castMedia = function(track) {
    if (!state.castSession) {
      showStatus('No hay sesión de Cast activa', 'error', 2000);
      return;
    }
    
    const mediaInfo = new chrome.cast.media.MediaInfo(track.url, 'audio/mp3');
    mediaInfo.metadata = new chrome.cast.media.MusicTrackMediaMetadata();
    mediaInfo.metadata.title = track.title;
    mediaInfo.metadata.artist = track.creator;
    mediaInfo.metadata.albumName = track.album || 'Unknown Album';
    
    if (track.id) {
      mediaInfo.metadata.images = [
        { url: `https://archive.org/services/img/${track.id}` }
      ];
    }
    
    const request = new chrome.cast.media.LoadRequest(mediaInfo);
    request.autoplay = true;
    request.currentTime = 0;
    
    state.castSession.loadMedia(request).then(
      () => {
        console.log('Media cargada en Cast');
        showStatus('Reproduciendo en dispositivo', 'info', 2000);
      },
      (error) => {
        console.error('Error cargando media en Cast:', error);
        showStatus('Error al reproducir en dispositivo', 'error', 2000);
      }
    );
  };
  
  console.log('Chromecast inicializado');
}

// ==================== NUEVO: FUNCIONALIDAD PWA ====================
function setupPWA() {
  // Registrar Service Worker
  if ('serviceWorker' in navigator) {
    window.addEventListener('load', () => {
      navigator.serviceWorker.register('/sw.js')
        .then((registration) => {
          console.log('ServiceWorker registrado exitosamente: ', registration.scope);
        })
        .catch((error) => {
          console.log('Error registrando ServiceWorker: ', error);
        });
    });
  }
  
  // Manejar instalación de PWA
  let deferredPrompt;
  
  window.addEventListener('beforeinstallprompt', (e) => {
    // Prevenir que el navegador muestre el prompt automáticamente
    e.preventDefault();
    // Guardar el evento para usarlo después
    deferredPrompt = e;
    
    // Opcional: Mostrar un botón de instalación personalizado
    showStatus('¡Puedes instalar esta aplicación!', 'info', 5000);
  });
  
  // Instalar PWA
  window.installPWA = function() {
    if (!deferredPrompt) {
      showStatus('La aplicación ya está instalada', 'info', 2000);
      return;
    }
    
    deferredPrompt.prompt();
    
    deferredPrompt.userChoice.then((choiceResult) => {
      if (choiceResult.outcome === 'accepted') {
        console.log('Usuario aceptó la instalación');
        showStatus('¡Aplicación instalada!', 'info', 3000);
      } else {
        console.log('Usuario rechazó la instalación');
      }
      deferredPrompt = null;
    });
  };
}

// ==================== FUNCIONALIDADES ESPECÍFICAS PARA ANDROID ====================
function setupAndroidFeatures() {
  if (!state.isAndroid) return;
  
  console.log('Inicializando funciones específicas para Android');
  
  // Mejorar compatibilidad con Android
  document.addEventListener('visibilitychange', () => {
    if (document.hidden) {
      // La aplicación está en segundo plano
      if (!elements.audio.paused) {
        console.log('App en segundo plano, manteniendo reproducción');
      }
    }
  });
  
  // Prevenir que la pantalla se apague durante la reproducción
  if ('wakeLock' in navigator) {
    let wakeLock = null;
    
    const requestWakeLock = async () => {
      try {
        wakeLock = await navigator.wakeLock.request('screen');
        console.log('Wake Lock activado para Android');
      } catch (err) {
        console.log('Wake Lock no disponible:', err);
      }
    };
    
    elements.audio.addEventListener('play', () => {
      requestWakeLock();
    });
    
    elements.audio.addEventListener('pause', () => {
      if (wakeLock) {
        wakeLock.release();
        wakeLock = null;
      }
    });
  }
  
  // Optimizar para pantallas táctiles
  document.documentElement.style.touchAction = 'manipulation';
  
  // Mejorar el feedback táctil
  const buttons = document.querySelectorAll('button');
  buttons.forEach(button => {
    button.addEventListener('touchstart', function() {
      this.style.transform = 'scale(0.95)';
    });
    
    button.addEventListener('touchend', function() {
      this.style.transform = 'scale(1)';
    });
  });
}

// ==================== MEJORAS PARA REPRODUCCIÓN EN SEGUNDO PLANO ====================
function setupVisibilityHandlers() {
  document.addEventListener('visibilitychange', handleVisibilityChange);
  document.addEventListener('pagehide', handlePageHide);
  document.addEventListener('pageshow', handlePageShow);
  window.addEventListener('blur', handleWindowBlur);
  window.addEventListener('focus', handleWindowFocus);
}

function handleVisibilityChange() {
  state.pageVisible = !document.hidden;
  
  if (state.pageVisible) {
    handlePageVisible();
  } else {
    handlePageHidden();
  }
}

function handlePageHide() {
  state.pageVisible = false;
  handlePageHidden();
}

function handlePageShow() {
  state.pageVisible = true;
  handlePageVisible();
}

function handleWindowBlur() {
  state.pageVisible = false;
  handlePageHidden();
}

function handleWindowFocus() {
  state.pageVisible = true;
  handlePageVisible();
}

function handlePageHidden() {
  console.log('Página en segundo plano - optimizando reproducción continua');
  
  if (state.backgroundPlaybackEnabled) {
    if (!elements.audio.paused && elements.audio.src) {
      showStatus('Reproduciendo en segundo plano...', 'info', 2000);
    }
  }
}

function handlePageVisible() {
  console.log('Página en primer plano - restaurando funcionalidad completa');
  
  if (state.currentTrack) {
    updateProgressBar();
  }
}

function setupEnhancedEndedHandling() {
  const audio = elements.audio;
  
  audio.addEventListener('ended', function() {
    console.log('Evento ended disparado - reproduciendo siguiente pista');
    
    const context = playbackManager.getCurrentContext();
    const albumContext = playbackManager.contextData.album;
    
    // Si estamos en un álbum de búsqueda y es la última pista
    if (context === 'album' && state.albumOrigin === 'search' && 
        albumContext.currentIndex >= albumContext.items.length - 1) {
      console.log('Last track of search album ended, playing next search result');
      // Pequeño delay para mejor experiencia de usuario
      setTimeout(() => playNextInContext(), 500);
    } else {
      // Comportamiento normal
      playNextInContext();
    }
  });
  
  let endedCheckInterval = null;
  
  audio.addEventListener('play', function() {
    if (endedCheckInterval) clearInterval(endedCheckInterval);
    
    endedCheckInterval = setInterval(() => {
      if (audio.duration > 0 && 
          audio.currentTime >= audio.duration - 0.5 && 
          !audio.paused) {
        console.log('Backup: Audio terminado, reproduciendo siguiente pista');
        clearInterval(endedCheckInterval);
        
        const context = playbackManager.getCurrentContext();
        const albumContext = playbackManager.contextData.album;
        
        if (context === 'album' && state.albumOrigin === 'search' && 
            albumContext.currentIndex >= albumContext.items.length - 1) {
          setTimeout(() => playNextInContext(), 500);
        } else {
          playNextInContext();
        }
      }
    }, 1000);
  });
  
  audio.addEventListener('pause', function() {
    if (endedCheckInterval) {
      clearInterval(endedCheckInterval);
      endedCheckInterval = null;
    }
  });
  
  audio.addEventListener('ended', function() {
    if (endedCheckInterval) {
      clearInterval(endedCheckInterval);
      endedCheckInterval = null;
    }
  });
}
